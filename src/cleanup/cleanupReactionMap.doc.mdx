# cleanupReactionMap

`cleanupReactionMap`
**AST Node Type:** *FunctionDeclaration*

> Creates a MobX reaction for Map structures with the setup/cleanup pattern: tracks key additions and removals while cleaning up previous effects.

#### Parameter:

- `expression`  — function that returns the Map or ObservableMap to observe.
- `effect`  — effect function for each new value (val, key, prev, map, r); returns a disposer.
- `opts`  — MobX reaction options.
#### Returns:
  for stopping the reaction and cleaning up all effects.

#### Source Code:

export function cleanupReactionMap<Key extends string | number | bigint = string, T = any, IMap extends Map<Key, T> = Map<Key, T>, FireImmediately extends boolean = true>(expression: (r: IReactionPublic) => IMap | ObservableMap<Key, T>, effect: (arg: T, key: Key, prev: FireImmediately extends true ? T | undefined : T, map: IMap, r: IReactionPublic) => ((isModified?: boolean) => void) | void, opts: IReactionOptions<T, FireImmediately> = {}): IReactionDisposer {
    const localDisposer = new Map<Key, ((isModified?: boolean) => void) | void>();
    let isInitialRun = true;
    const disposer = cleanupReaction<MapSnapshot<Key, T, IMap>>((r) => {
        const map = expression(r);
        return {
            keys: [...map.keys()],
            map,
        };
    }, (arg) => {
        for (const key of localDisposer.keys()) {
            if (!arg.map.has(key)) {
                localDisposer.get(key)?.();
                localDisposer.delete(key);
            }
        }
        for (const key of arg.keys) {
            if (!localDisposer.has(key)) {
                localDisposer.set(key, cleanupReaction((r) => {
                    return expression(r).get(key)!;
                }, (v, prev, r) => {
                    return effect(v, key, prev, expression(r), r);
                }, isInitialRun ? opts : ({ ...opts, fireImmediately: true } as typeof opts)));
            }
        }
        isInitialRun = false;
    }, {
        fireImmediately: true,
        equals: hasSameKeys,
    });
    const newDisposer: IReactionDisposer = () => {
        disposer();
        localDisposer.forEach((v) => {
            v?.();
        });
    };
    newDisposer.$mobx = disposer.$mobx;
    return newDisposer;
}
